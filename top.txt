function bindNav(){
  document.querySelectorAll('.nav-btn,[data-page]').forEach(btn=>{
    btn.onclick=(e)=>{
      const key=btn.dataset.page;
      if(!key)return;
      e.preventDefault(); e.stopPropagation();
      app.page=key;
      renderPage();
      document.querySelectorAll('.nav-btn').forEach(x=>x.classList.toggle('active',x.dataset.page===key));
      window.scrollTo({top:0,behavior:'smooth'});
    };
  });
  if($('bizSwitch'))$('bizSwitch').onclick=(e)=>{e.preventDefault();app.business=null;app.membership=null;app.data=null;app.page='dashboard';render()};
  if($('logoutTop'))$('logoutTop').onclick=async(e)=>{e.preventDefault();await logout()};
}

function reportData(shiftOverride=null){
  const d=app.data,day=today(),scope=shiftOverride||null;
  const filterShift=x=>!scope||recordShift(x)===scope;
  const sales=d.sales.filter(x=>x.date===day&&filterShift(x)),exp=d.expenses.filter(x=>x.date===day&&filterShift(x)),deps=d.deposits.filter(x=>x.date===day&&filterShift(x));
  const byCh={}; for(const x of revenueChannels())byCh[x.key]=[0,0,x]; sales.forEach(x=>{if(byCh[x.channel])byCh[x.channel][x.currency==='KHR'?0:1]+=x.amount});
  const paymentRows={Cash:{k:0,u:0},ABA:{k:0,u:0},Other:{k:0,u:0}}; sales.forEach(x=>{const key=paymentRows[x.payment]?x.payment:'Other';paymentRows[key][x.currency==='KHR'?'k':'u']+=x.amount});
  const totalK=sales.filter(x=>x.currency==='KHR').reduce((a,x)=>a+x.amount,0),totalU=sales.filter(x=>x.currency==='USD').reduce((a,x)=>a+x.amount,0);
  const expK=exp.filter(x=>x.currency==='KHR').reduce((a,x)=>a+x.amount,0),expU=exp.filter(x=>x.currency==='USD').reduce((a,x)=>a+x.amount,0);
  const depK=deps.filter(x=>x.currency==='KHR').reduce((a,x)=>a+x.amount,0),depU=deps.filter(x=>x.currency==='USD').reduce((a,x)=>a+x.amount,0);
  const cups=sales.reduce((a,x)=>a+Number(x.cups||0),0);
  const itemMap={}; sales.forEach(s=>s.items?.forEach(i=>{const key=i.en||i.kh||i.id||'—';if(!itemMap[key])itemMap[key]={en:i.en||key,kh:i.kh||i.en||key,qty:0,amountK:0,amountU:0};itemMap[key].qty+=Number(i.qty)||0;const itemAmount=(Number(i.price)||0)*(Number(i.qty)||0);if(s.currency==='USD')itemMap[key].amountU+=itemAmount;else itemMap[key].amountK+=itemAmount}));
  const topItems=Object.values(itemMap).sort((a,b)=>b.qty-a.qty||((b.amountK+b.amountU)-(a.amountK+a.amountU))).slice(0,3),best=topItems[0]?.[app.lang]||topItems[0]?.en||'—';
  const promoCounts={};sales.forEach(s=>{if(s.promotionId){const p=d.promotions.find(x=>x.id===s.promotionId);const name=p?.[app.lang]||p?.en||s.promotionId;promoCounts[name]=(promoCounts[name]||0)+1;}});const promotionSummary=Object.entries(promoCounts).sort((a,b)=>b[1]-a[1])[0]||null;
  const allCounts=d.cashCounts.filter(x=>x.date===day).sort((a,b)=>String(a.time||'').localeCompare(String(b.time||'')));
  let prev=null,actual=null;
  if(scope==='morning'){const prior=d.cashCounts.filter(x=>x.date<day).sort((a,b)=>String(a.date+' '+a.time).localeCompare(String(b.date+' '+b.time)));prev=prior[prior.length-1]||null;actual=allCounts.filter(x=>recordShift(x)==='morning').slice(-1)[0]||null;}
  else if(scope==='day'){prev=allCounts.filter(x=>recordShift(x)==='morning').slice(-1)[0]||null;actual=allCounts.filter(x=>recordShift(x)==='day').slice(-1)[0]||null;}
  else {const prior=d.cashCounts.filter(x=>x.date<day).sort((a,b)=>String(a.date+' '+a.time).localeCompare(String(b.date+' '+b.time)));prev=prior[prior.length-1]||null;actual=allCounts[allCounts.length-1]||null;}
  const prevK=Number(prev?.actualKhr)||0,prevU=Number(prev?.actualUsd)||0;
  const csK=sales.filter(x=>x.payment==='Cash'&&x.currency==='KHR').reduce((a,x)=>a+x.amount,0),csU=sales.filter(x=>x.payment==='Cash'&&x.currency==='USD').reduce((a,x)=>a+x.amount,0);
  const ceK=exp.filter(x=>x.method==='cash'&&x.currency==='KHR').reduce((a,x)=>a+x.amount,0),ceU=exp.filter(x=>x.method==='cash'&&x.currency==='USD').reduce((a,x)=>a+x.amount,0);
  return {byCh,totalK,totalU,expK,expU,depK,depU,cups,best,prevK,prevU,csK,csU,ceK,ceU,expectedK:prevK+csK-ceK-depK,expectedU:prevU+csU-ceU-depU,actualK:actual?.actualKhr??null,actualU:actual?.actualUsd??null,paymentRows,topItems,promotionSummary};
}
function getReportTemplate(){

  merged.telegramTemplateEn=String(x.telegramTemplateEn||base.telegramTemplateEn||'').trim()||base.telegramTemplateEn;
  return merged;
}

function buildReportTextPreview(){
  const rt=getReportTemplate(),r=reportData(),m=senderInfo(),rows=[];
  if(rt.header) rows.push(rt.header); rows.push(t('report')+' · '+today());
  if(rt.showDate) rows.push((app.lang==='kh'?'កាលបរិច្ឆេទ':'Date')+': '+today());
  if(rt.showStaff) rows.push(t('sender')+': '+m.name);
  if(rt.sections.summary) rows.push(`${t('cups')}: ${r.cups} · ${t('revenue')}: ${money(r.totalK)} · ${money(r.totalU,'USD')}`);
  if(rt.sections.payments) rows.push(`${t('payments')}: Cash ${money(r.paymentRows.Cash.k)} · ABA ${money(r.paymentRows.ABA.k)} · Other ${money(r.paymentRows.Other.k)}`);
  if(rt.sections.expenses) rows.push(`${t('expense')}: ${money(r.expK)} · ${money(r.expU,'USD')}`);
  if(rt.sections.cash) rows.push(`${t('expected')}: ${money(r.expectedK)} · ${money(r.expectedU,'USD')}`);
  if(rt.sections.products) rows.push(`${t('bestSeller')}: ${r.best||'—'}`);
  if(rt.showRate) rows.push(`${t('rate')}: 1 USD = ${Number(app.data.settings.rate||0).toLocaleString()} KHR`);
  if(rt.showSender) rows.push(`${t('sender')}: ${m.name} · #${m.shortId}`);
  if(rt.footer) rows.push(rt.footer);
  return `<pre>${esc(rows.join('\n'))}</pre>`;
}
